# Menú Familiar V2.0.0 release gate

The upgrade applies migration 0011 to a production snapshot and imports missing approved recipes and photos. It preserves all dish names, including the two proposed renames in QA, and keeps existing drafts, publications and photo references. Verified source links take precedence. A V1 row can acquire a link only if the owner confirms that the original CSV seeded production and the row matches its seed ID, original name, categories, meal preference and occasion flag. Missing or ambiguous identities are reported and skipped. Publishing updates affected editable recipe snapshots and undo state. Past snapshots, menu choices, templates and V1 shopping records are preserved.

~~~bash
prod() {
    docker compose --project-name menu-familiar --env-file "$repo/.env" -f "$repo/compose.yaml" "$@"

manage() {
    local directory=$1
    shift
    DJANGO_ENV=development DJANGO_DEBUG=0 DJANGO_ALLOWED_HOSTS=localhost \
        DJANGO_DB_PATH="$directory/menu.sqlite3" DJANGO_RECIPE_PHOTO_ROOT="$directory/recipe-photos" \
        "$py" "$repo/manage.py" "$@"
}
import_recipes() {
    local directory=$1
    shift
    manage "$directory" import_recipe_proposals "$bundle" "${legacy_opt[@]}" --preserve-dish-names "$@"
}
madrid_week() {
    "$py" -c 'from datetime import datetime,timedelta; from zoneinfo import ZoneInfo; d=datetime.now(ZoneInfo("Europe/Madrid")).date(); print(d-timedelta(days=d.weekday()))'
}
snapshot() {
    manage "$1" backup_menu_data "$2"
}
stage() {
    cp -a "$1" "$2"
    manage "$2" migrate --noinput
    manage "$2" migrate --check
    manage "$2" check
}
compare_apply() {
    diff -u <(tail -n +2 "$1") <(tail -n +2 "$2")
}
smoke() {
    test "$(curl --noproxy '*' -fsS --retry 10 --retry-connrefused --retry-delay 1 --max-time 5 "$origin/healthz")" = OK
    local path
    for path in /menu/ /plats/ /compra/ /static/menus/app.css "$@"; do
        curl --noproxy '*' -fsS --max-time 5 -o /dev/null "$origin$path"
    done
}
photo_route() {
    "$py" - "$1/menu.sqlite3" <<'PY'
from pathlib import Path
import sqlite3
import sys
with sqlite3.connect(Path(sys.argv[1]).as_uri() + "?mode=ro", uri=True) as db:
    row = db.execute("SELECT dish_id FROM menus_cataloguesourcerecord WHERE photo_digest != '' ORDER BY source_id LIMIT 1").fetchone()
    if row:
        print(f"/plats/{row[0]}/recepta/foto/")
PY
}
~~~

~~~bash
# The authorized host operator assigns only the disposable test directories.
verify_image() (
    local image_ref=$1 directory=$2 route=${3:-}
    chown -R 10001:10001 "$directory"
    local container
    container=$(docker run -d --network none --mount "type=bind,src=$directory,dst=/data" \
        -e DJANGO_SECRET_KEY=isolated-menu-release-rehearsal -e DJANGO_ALLOWED_HOSTS=127.0.0.1 \
        -e DJANGO_DB_PATH=/data/menu.sqlite3 "$image_ref")
    trap 'docker rm -f "$container" >/dev/null' EXIT
    docker exec "$container" python manage.py migrate --check
    docker exec -i "$container" python - "$route" <<'PY'
import sys
import time
from urllib.request import urlopen
origin = "http://127.0.0.1:8000"
for _ in range(30):
    try:
        with urlopen(origin + "/healthz", timeout=2) as response:
            if response.read() == b"OK":
                break
    except OSError:
        pass
    time.sleep(1)
else:
    raise SystemExit("Health check timed out")
for route in filter(None, ("/menu/", "/plats/", "/compra/", "/static/menus/app.css", sys.argv[1])):
    with urlopen(origin + route, timeout=5) as response:
        if response.status != 200 or not response.read():
            raise SystemExit(f"Smoke check failed for {route}")
PY
)
verify_image "$old_image" "$work/rehearsal-v1-restore"
verify_image "$image_id" "$work/rehearsal-stage" "$(photo_route "$work/rehearsal-stage")"
~~~

## Owner-authorized cutover

Pause household edits through confirmation or rollback and avoid the Madrid Monday rollover. Allow a few minutes of downtime provisionally and use the timed production rehearsal to set the window. The outage begins at the service stop. If any subsequent command fails, use the rollback block from a shell with the recorded variables.

~~~bash
cmp "$repo/.env" "$work/v1.env"
cmp "$repo/compose.yaml" "$work/v1.compose.yaml"
test "$(prod ps -q menu-familiar)" = "$old_container"
test "$(docker inspect --format '{{.Image}}' "$old_container")" = "$old_image"
test "$(docker image inspect --format '{{.Id}}' "$image")" = "$image_id"
test "$(docker image inspect --format '{{.Id}}' "$old_tag")" = "$old_image"
original="$data.v1.$(basename "$work")"
~~~

## Rollback

Use the untouched original directory and the recorded V1 image. If the cutover failed before moving the original, the original is still at the production path. Preserve any failed V2 directory, including any household edits that reached V2, for separate reconciliation.

~~~bash
prod stop menu-familiar
if test -d "$original"; then
    failed="$data.failed-v2.$(basename "$work")"
    test ! -e "$failed" && test ! -L "$failed" || exit 1
    if test -d "$data"; then mv -T "$data" "$failed"; fi
    mv -T "$original" "$data"
fi
test -f "$data/menu.sqlite3"
rollback_env=$(mktemp "$repo/.menu-v1.env.XXXXXXXX")
install -m 0600 "$work/v1.env" "$rollback_env"
mv -T "$rollback_env" "$repo/.env"
cmp "$repo/compose.yaml" "$work/v1.compose.yaml"
if ! docker image inspect "$old_tag" >/dev/null 2>&1; then
    docker image load --input "$work/v1-image.tar"
fi
test "$(docker image inspect --format '{{.Id}}' "$old_tag")" = "$old_image"
prod up -d --no-build --pull never menu-familiar
test "$(docker inspect --format '{{.Image}}' "$(prod ps -q menu-familiar)")" = "$old_image"
test "$(docker port "$(prod ps -q menu-familiar)" 8000/tcp)" = "$binding"
smoke
~~~
